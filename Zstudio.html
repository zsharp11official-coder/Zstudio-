class TabletCanvasApp {
    fn init(initialBrushSize = 5.0) {
        self.brushSize = initialBrushSize
        self.activeColor = "#e11d48"
    }

    # Native Tablet SplitView Layout Primitive
    fn render() {
        return SplitView(
            sidebar: ToolPanel(
                brushSize: self.brushSize,
                color: self.activeColor
            ),
            main: DrawingCanvas(
                palmRejection: true,
                strokeColor: self.activeColor,
                onStylusDraw: (stroke) => self.processStroke(stroke)
            )
        )
    }

    # Functional Pipeline Operator (|>) for Pressure Stream Smoothing
    fn processStroke(stroke) {
        return stroke
            |> filterPalmInputs
            |> scalePressure(multiplier: 1.8)
            |> smoothBezierCurves
    }
}
